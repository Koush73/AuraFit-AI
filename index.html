-- Production DDL Schema CREATE TABLE users ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), email VARCHAR(255) UNIQUE NOT NULL, hashed_password VARCHAR(255) NOT NULL, full_name VARCHAR(100), is_active BOOLEAN DEFAULT TRUE, created_at TIMESTAMP DEFAULT NOW() ); CREATE TABLE user_profiles ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES users(id) ON DELETE CASCADE, age INT, gender VARCHAR(10), height_cm NUMERIC, weight_kg NUMERIC, training_experience VARCHAR(20), fitness_goal VARCHAR(30), duration_mins INT, days_per_week INT, equipment VARCHAR(30), dietary_preference VARCHAR(30), updated_at TIMESTAMP DEFAULT NOW() ); CREATE TABLE physique_scans ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES users(id) ON DELETE CASCADE, current_img_url TEXT, target_img_url TEXT, current_bf_pct NUMERIC, target_bf_pct NUMERIC, v_taper_ratio NUMERIC, morphological_delta JSONB, scanned_at TIMESTAMP DEFAULT NOW() ); CREATE TABLE progress_logs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES users(id) ON DELETE CASCADE, weight_kg NUMERIC, waist_cm NUMERIC, chest_cm NUMERIC, arm_cm NUMERIC, adherence_rpe VARCHAR(20), adaptive_action_taken TEXT, logged_at TIMESTAMP DEFAULT NOW() );
